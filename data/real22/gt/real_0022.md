void Plane::setTimer() {
//系统自带设定定时器的函数
::SetTimer(m_hwnd, 1/*定时器的id*/, 80/*触发的频率，毫秒*/, nullptr);
//同时按下多个方向键，只读取最后一个按下的
//优化：定时检测方向键是否按下，如果按下的话，在按照设定频率去移动
::SetTimer(m_hwnd, 2/*定时器的id*/, 20/*触发的频率，毫秒*/, nullptr);
//炮弹定时移动
::SetTimer(m_hwnd, 3/*定时器的id*/, 60/*触发的频率，毫秒*/, nullptr);
//敌人飞机定时移动
::SetTimer(m_hwnd, 4/*定时器的id*/, 60/*触发的频率，毫秒*/, nullptr);
//敌人飞机定时创建
::SetTimer(m_hwnd, 5/*定时器的id*/, 3000/*触发的频率，毫秒*/, nullptr);
//判断是否碰撞（炮弹和敌人飞机，玩家飞机和敌人飞机）
::SetTimer(m_hwnd, 6/*定时器的id*/, 10/*触发的频率，毫秒*/, nullptr);
//定时切换爆炸效果
::SetTimer(m_hwnd, 7/*定时器的id*/, 200/*触发的频率，毫秒*/, nullptr);
}
void Plane::killTimer()
{
::KillTimer(m_hwnd, 1/*定时器的id*/);
::KillTimer(m_hwnd, 2/*定时器的id*/);
::KillTimer(m_hwnd, 3/*定时器的id*/);
::KillTimer(m_hwnd, 4/*定时器的id*/);
::KillTimer(m_hwnd, 5/*定时器的id*/);
::KillTimer(m_hwnd, 6/*定时器的id*/);
::KillTimer(m_hwnd, 7/*定时器的id*/);
}
void Plane::showScore() {
//显示分数
//整型转字符串。
WCHAR sc[5] = { 0 };
_itow_s(m_score, sc, 10);

RECT rect{ 0, 0, 120, 40 };
drawtext(sc, &rect, DT_CENTER | DT_VCENTER | DT_SINGLELINE);
settextcolor(RGB(200, 0, 0));
}
